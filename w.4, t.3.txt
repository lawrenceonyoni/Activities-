#include <iostream>
#include <cstdlib> // For rand() and srand()
#include <ctime>   // For time()

using namespace std;

int main() {
    // Seed the random number generator using the current time
    srand(time(0));

    // =========================================================================
    // 1D Array – Weekly Revenue Tracker
    // =========================================================================
    cout << "=== 1D ARRAY: WEEKLY REVENUE TRACKER ===" << endl;
    
    // 1. Use a 1D array revenue[7] to store the daily revenues
    double revenue[7];
    double totalRevenue = 0.0;
    double averageRevenue = 0.0;
    
    // 2. Input the revenue for each day
    string days[7] = {"Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"};
    for (int i = 0; i < 7; i++) {
        cout << "Enter revenue for " << days[i] << ": ";
        cin >> revenue[i];
        
        // 3. Calculate the total weekly revenue
        totalRevenue += revenue[i];
    }
    
    // 3. Calculate the average daily revenue
    averageRevenue = totalRevenue / 7.0;
    
    // Display 1D Array results
    cout << "\n--- Revenue Summary ---" << endl;
    cout << "Total Weekly Revenue : " << totalRevenue << endl;
    cout << "Average Daily Revenue: " << averageRevenue << endl;
    cout << "========================================" << endl << endl;


    // =========================================================================
    // 2D Array – Room Occupancy (One Branch)
    // =========================================================================
    cout << "=== 2D ARRAY: ROOM OCCUPANCY (ONE BRANCH) ===" << endl;
    
    // 1. Use a 2D array occupancy[5][10] (5 floors, 10 rooms per floor)
    int occupancy[5][10];
    
    // 2. Simulate random input (0 or 1)
    for (int floor = 0; floor < 5; floor++) {
        for (int room = 0; room < 10; room++) {
            occupancy[floor][room] = rand() % 2; // Generates either 0 or 1
        }
    }
    
    // 3. Display the number of occupied and vacant rooms per floor
    for (int floor = 0; floor < 5; floor++) {
        int occupiedCount = 0;
        int vacantCount = 0;
        
        for (int room = 0; room < 10; room++) {
            if (occupancy[floor][room] == 1) {
                occupiedCount++;
            } else {
                vacantCount++;
            }
        }
        cout << "Floor " << (floor + 1) << " -> Occupied Rooms: " << occupiedCount 
             << " | Vacant Rooms: " << vacantCount << endl;
    }
    cout << "=============================================" << endl << endl;


    // =========================================================================
    // 3D Array – Multiple Branches
    // =========================================================================
    cout << "=== 3D ARRAY: MULTIPLE BRANCHES ===" << endl;
    
    // 1. Use a 3D array chain[3][5][10] to represent 3 branches
    int chain[3][5][10];
    int totalChainOccupied = 0;
    
    // 2. Assign random occupancy (1 or 0) to each room across all dimensions
    for (int branch = 0; branch < 3; branch++) {
        for (int floor = 0; floor < 5; floor++) {
            for (int room = 0; room < 10; room++) {
                chain[branch][floor][room] = rand() % 2;
                
                // 3. Calculate total number of occupied rooms
                if (chain[branch][floor][room] == 1) {
                    totalChainOccupied++;
                }
            }
        }
    }
    
    // 3. Display the total number of occupied rooms across all branches
    cout << "Total Occupied Rooms Across All 3 Branches: " << totalChainOccupied << " / 150 rooms." << endl;
    cout << "============================================" << endl;

    return 0;
}
