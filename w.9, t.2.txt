// iii) Create a derived class named Car that inherits from Vehicle
class Car : public Vehicle {
private:
    int numberOfDoors;
    double engineCapacity; // e.g., 1.5, 2.0 liters

public:
    // Input function combining base and car-specific logic
    void inputCarDetails() {
        inputVehicle(); // Call base class input
        cout << "Enter Number of Doors: ";
        cin >> numberOfDoors;
        cout << "Enter Engine Capacity (Liters): ";
        cin >> engineCapacity;
    }

    // iv) Include a member function in the derived class
    void displayCar() {
        cout << "\n--- Car Specifications Report ---" << endl;
        displayVehicle(); // Call ii) displayVehicle() for inherited attributes
        cout << "Number of Doors:  " << numberOfDoors << endl;
        cout << "Engine Capacity:  " << engineCapacity << "L" << endl;
        cout << "---------------------------------" << endl;
    }
};

// v) In the main function
int main() {
    cout << "\n=== Transport Company Vehicle Registry ===" << endl;
    
    // Create an object of the Car class
    Car companyCar;

    // Input all details
